SELECT SUM(dbo.[1TabPeças].PrVenda) AS PrVenda, SUM(dbo.METAIS.OuroQtde) AS Au, SUM(dbo.[1TabPeças].Custo) AS Custo, COUNT(dbo.[1TabPeças].IDPeça) AS Qtd FROM VENDAS RIGHT OUTER JOIN [VENDAS DET] ON dbo.VENDAS.Id = dbo.[VENDAS DET].Venda RIGHT OUTER JOIN [1TabPeças] LEFT OUTER JOIN METAIS ON dbo.[1TabPeças].IDPeça = dbo.METAIS.Id INNER JOIN PEENBase ON dbo.[1TabPeças].IDPeça = dbo.PEENBase.IDPeça AND dbo.[1TabPeças].IDPeça = dbo.PEENBase.IDPeça ON dbo.[VENDAS DET].Peça = dbo.[1TabPeças].IDPeça WHERE (dbo.VENDAS.DataVenda IS NULL) AND (dbo.[1TabPeças].Vendida = 0 OR dbo.[1TabPeças].Vendida IS NULL) AND (dbo.[1TabPeças].Modêlo IS NOT NULL)
